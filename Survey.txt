        // Add card to gallery
        function addToGallery(cardId) {
            const card = cardsData.find(c => c.id === cardId);
            if (!card) return;
            
            // Check if card is already in gallery
            if (myGallery.some(item => item.id === cardId)) {
                showToast('This card is already in your gallery!', 'warning');
                return;
            }
            
            // Add card to gallery
            myGallery.push(card);
            localStorage.setItem('myGallery', JSON.stringify(myGallery));
            
            showToast('Card added to your gallery!', 'success');
            
            // Update button text
            const button = event.target;
            button.innerHTML = '<i class="fas fa-images mr-2"></i>Already in Gallery';
            button.disabled = true;
            button.classList.add('opacity-75', 'cursor-not-allowed');
        }

        // Remove card from gallery
        function removeFromGallery(event, cardId) {
            event.stopPropagation();
            
            // Remove card from gallery
            myGallery = myGallery.filter(item => item.id !== cardId);
            localStorage.setItem('myGallery', JSON.stringify(myGallery));
            
            showToast('Card removed from your gallery', 'info');
            
            // Regenerate my gallery
            generateMyGallery();
            
            // Regenerate main gallery to update the "Saved" status
            generateCards();
        }

        // Share card via WhatsApp
        function shareViaWhatsApp(cardId) {
            const card = cardsData.find(c => c.id === cardId);
            if (!card) return;
            
            // Create the message to share
            const message = `*${card.title}*\n\n${card.description}\n\nCard Code: ${card.code}\n\nCheck out this card: ${window.location.href}`;
            
            // Encode the message for URL
            const encodedMessage = encodeURIComponent(message);
            
            // Create WhatsApp URL
            const whatsappUrl = `https://wa.me/?text=${encodedMessage}`;
            
            // Open WhatsApp in a new tab
            window.open(whatsappUrl, '_blank');
            
            showToast('Opening WhatsApp to share this card...', 'success');
        }

        // Edit card
        function editCard(cardId) {
            const card = cardsData.find(c => c.id === cardId);
            if (!card) return;
            
            currentEditingCard = card;
            
            // Populate form fields
            document.getElementById('editTitle').value = card.title;
            document.getElementById('editCategory').value = card.category;
            document.getElementById('editDescription').value = card.description;
            document.getElementById('editPrice').value = card.price;
            document.getElementById('editImage').value = card.image;
            document.getElementById('cardCodeDisplay').textContent = card.code;
            
            // Initialize CodeMirror if not already done
            if (!codeEditor) {
                codeEditor = CodeMirror.fromTextArea(document.getElementById('codeEditor'), {
                    mode: 'javascript',
                    theme: 'monokai',
                    lineNumbers: true,
                    autoCloseBrackets: true,
                    matchBrackets: true,
                    indentUnit: 2,
                    tabSize: 2,
                    lineWrapping: true
                });
            }
            
            // Set JSON data in code editor
            const cardData = {
                id: card.id,
                title: card.title,
                category: card.category,
                description: card.description,
                price: card.price,
                image: card.image,
                code: card.code
            };
            codeEditor.setValue(JSON.stringify(cardData, null, 2));
            
            // Show modal
            document.getElementById('editModal').classList.remove('hidden');
            
            // Set active tab to form view
            setActiveTab('formTab', 'formView');
        }

        // Close edit modal
        function closeEditModal() {
            document.getElementById('editModal').classList.add('hidden');
            currentEditingCard = null;
        }

        // Generate new code
        function generateNewCode() {
            const characters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
            let result = '';
            const length = 4;
            
            for (let i = 0; i < length; i++) {
                result += characters.charAt(Math.floor(Math.random() * characters.length));
            }
            
            const newCode = `CARD-${currentEditingCard.id}-${result}`;
            document.getElementById('cardCodeDisplay').textContent = newCode;
            
            // Update code editor if visible
            if (document.getElementById('codeView').classList.contains('hidden') === false) {
                try {
                    const cardData = JSON.parse(codeEditor.getValue());
                    cardData.code = newCode;
                    codeEditor.setValue(JSON.stringify(cardData, null, 2));
                } catch (e) {
                    // Ignore if JSON is invalid
                }
            }
        }

        // Format JSON in code editor

            }
        }

            document.getElementById(tabId).classList.remove('text-gray-500');
            document.getElementById(tabId).classList.add('text-purple-600', 'border-b-2', 'border-purple-600');

            document.getElementById('formView').classList.add('hidden');
            document.getElementById('codeView').classList.add('hidden');
            document.getElementById(viewId).classList.remove('hidden');
        }

        // Save card changes
        function saveCardChanges() {
            if (!currentEditingCard) return;
            
            let cardData;
            
            // Check which view is active
            if (document.getElementById('formView').classList.contains('hidden') === false) {
                // Form view is active
                cardData = {
                    id: currentEditingCard.id,
                    title: document.getElementById('editTitle').value,
                    category: document.getElementById('editCategory').value,
                    description: document.getElementById('editDescription').value,
                    price: document.getElementById('editPrice').value,
                    image: document.getElementById('editImage').value,
                    code: document.getElementById('cardCodeDisplay').textContent
                };
            } else {
                // Code view is active
                try {
                    cardData = JSON.parse(codeEditor.getValue());
                    
                    // Update form fields with values from code editor
                    document.getElementById('editTitle').value = cardData.title;
                    document.getElementById('editCategory').value = cardData.category;
                    document.getElementById('editDescription').value = cardData.description;
                    document.getElementById('editPrice').value = cardData.price;
                    document.getElementById('editImage').value = cardData.image;
                    document.getElementById('cardCodeDisplay').textContent = cardData.code;
                } catch (e) {
                    showToast('Invalid JSON: ' + e.message, 'error');
                    return;
                }
            }
            
            // Update card data
            currentEditingCard.title = cardData.title;
            currentEditingCard.category = cardData.category;
            currentEditingCard.description = cardData.description;
            currentEditingCard.price = cardData.price;
            currentEditingCard.image = cardData.image;
            currentEditingCard.code = cardData.code;
            
            // Update in myGallery if present
            const myGalleryIndex = myGallery.findIndex(item => item.id === currentEditingCard.id);
            if (myGalleryIndex !== -1) {
                myGallery[myGalleryIndex] = {...currentEditingCard};
                localStorage.setItem('myGallery', JSON.stringify(myGallery));
            }
            
            // Close modal
            closeEditModal();
            
            // Refresh detail view
            showDetail(currentEditingCard.id);
            
            showToast('Card updated successfully!', 'success');
        }

        // Show toast notification
        function showToast(message, type = 'success') {
            const toast = document.getElementById('toast');
            toast.textContent = message;
            
            // Set background color based on type
            switch(type) {
                case 'success':
                    toast.style.backgroundColor = '#10B981'; // green
                    break;
                case 'warning':
                    toast.style.backgroundColor = '#F59E0B'; // yellow
                    break;
                case 'error':
                    toast.style.backgroundColor = '#EF4444'; // red
                    break;
                case 'info':
                    toast.style.backgroundColor = '#3B82F6'; // blue
                    break;
                default:
                    toast.style.backgroundColor = '#4F46E5'; // purple
            }
            
            // Show toast
            toast.classList.add('show');
            
            // Hide toast after 3 seconds
            setTimeout(() => {
                toast.classList.remove('show');
            }, 3000);
        }
